cmake_minimum_required(VERSION 3.16)
project(paping VERSION 2.0.0 LANGUAGES CXX)

add_executable(paping src/main.cpp src/net.cpp)
target_compile_features(paping PRIVATE cxx_std_17)
set_target_properties(paping PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(paping PRIVATE /W4 /permissive- /utf-8)
    set_property(TARGET paping PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(paping PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(WIN32)
    target_link_libraries(paping PRIVATE ws2_32)
endif()
find_package(Threads REQUIRED)
target_link_libraries(paping PRIVATE Threads::Threads)
include(GNUInstallDirs)
install(TARGETS paping RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:paping>)
    set_tests_properties(integration PROPERTIES TIMEOUT 90)
    add_executable(stats_test tests/stats.cpp)
    target_compile_features(stats_test PRIVATE cxx_std_17)
    add_test(NAME statistics COMMAND stats_test)
endif()
